**#1楼**

大半年前做的东西，当时做得很简陋，可能有各种bug，轻喷~
起因是今天有朋友在问怎么往wz里添加新的装备，因为自己想做一套怪怪卡收集/矿戒的玩法。我的观点是比起编辑wz的复杂操作，用wz里已有的装备加上定义好的javascript api，只要用txt编辑器就能把这个需求实现了，更简单。所以想分享这个。
出处
这个脚本是被我放在我自己的仓库
亚马乌罗提MS
里的1002006.js(在明珠港右上角找到吉夫)。【我当时很不会用git（现在也很不会），所以直接把自己的本地文件全部上传了，没有用网站自带的fork，不要骂我...】
功能简介
所实现的功能是：在完成特定的任务之后，可以来吉夫这里领取可以让自己自行分配属性点的戒指，玩家如果缺少血量，这个戒指就可以多加点血，如果本身血比较多，就可以因此获取比较多的攻击。
这个戒指呢，在玩家等级比较高的时候可以更新（就已经写好的内容而言，玩家在30级之后做完桑拿服任务可以领取“一克拉戒指”，在70级做完骑宠任务可以换成“二克拉戒指”，后续内容没放到仓库里，可以自己更新...）
开发的想法
设计的初衷是希望玩家去做有趣的任务，同时给玩家“做有意义的选择”。
“游戏的本质就是要让玩家做有意义的选择”——文鸽
分享的意图
半年前写的，内容很简陋，跑起来说不定还有什么漏洞。但是想借此分享一下自己对MMORPG玩法的一些理解，顺便帮助朋友弄明白CharacterExtendValue这一系列API（北斗附带的，让数据库里给一个角色存取特定的信息）还有gainEquip这个API（发放装备并且赋予装备自定义的属性）怎么使用。
谢谢有耐心阅读完这些文字的朋友。

var status = -1;
var freePoint = 0;
var addAbility = 0;
var addHp = 0;
var addMp = 0;
var addATK = 0;
var addMATK = 0;
var RingFlag = 0;

function action(mode, type, selection) {
    status++;
    if (mode != 1) {
        if (mode == 0) {
            cm.sendOk("如果你想获得#b闪耀新星戒指#k，随时可以来找我哦。");
        }
        cm.dispose();
        return;
    }
    if (status == 0) {
        cm.sendYesNo("你想获得#b闪耀新星戒指#k吗？不过我不想轻易地送给您。您先前往#b林中之城#k找到#b泰素夫#k，执行完#b泰素夫的秘密之书#k这个任务以后，我就把最基础的#b闪耀新星戒指1克拉#k送给您。以后您完成了指定的更多任务，我还会给您更强的别的戒指。");
    } else if (status == 1) {
        let RingFlagString = cm.getCharacterExtendValue("闪耀新星戒指克拉数");
        if (RingFlagString == null) {
            cm.saveOrUpdateCharacterExtendValue("闪耀新星戒指克拉数", "0");
            RingFlag = 0;
        } else {
            RingFlag = parseInt(RingFlagString);
        }
        if (cm.isQuestCompleted(2023) && (RingFlag < 2)) {
            let text = "你现在完成了#b泰素夫的秘密之书#k，我可以送你#b闪耀新星戒指1克拉#k，你是需要：#b\r\n";
            text += "#L0#我现在没有戒指，想要一个新的#l\r\n";
            text += "#L1#我想要重新分配这个戒指的属性#l\r\n";
            text += "#L2#我想要2克拉的戒指#l\r\n";
            cm.sendSimple(text);
        } else if (!cm.isQuestCompleted(2023)){
            cm.sendOk("您执行了#b泰素夫的秘密之书#k这个任务以后再来找我吧，是那个获取#b桑拿服#k的任务。");
            cm.dispose();
        } else if (RingFlag == 2) {
            let text = "你现在完成了#b泰素夫的秘密之书#k和#b费洛蒙香水#k，我可以为你调整#b闪耀新星戒指2克拉#k的属性，你是需要：#b\r\n";
            text += "#L0#我想要重新分配这个戒指的属性#l\r\n";
            text += "#L1#我想要3克拉的戒指#l\r\n";
            cm.sendSimple(text);
        }
    } else if (status == 2) {
        if (cm.isQuestCompleted(2023) && (RingFlag < 2)) {
            if (selection == 0) {
                if (!cm.haveItem(1112303) && (RingFlag < 1)) {
                    cm.getPlayer().gainEquip(1112303, 4, 4, 4, 4, 1000, 1000, 4, 12, 0, 0, 0, 0, 0, 0, 0, 0, -1);
                    cm.saveOrUpdateCharacterExtendValue("闪耀新星戒指克拉数", "1");
                    cm.sendOk("搞定！");
                    cm.dispose();
                } else {
                    cm.sendOk("不许多拿！");
                    cm.dispose();
                }
                
            } else if (selection == 1) {
                if (cm.haveItem(1112303)) {
                    freePoint = 20;
                    cm.saveOrUpdateCharacterExtendValue("闪耀新星戒指克拉数", "1");
                    RingFlag = 1;
                    cm.sendGetNumber("#b闪耀新星戒指1克拉#k一共有#b20点#k可分配属性。每点可分配属性可以变成#b1#k四维、#b250#kHP、#b250#kMP、#b1#k攻击或者#b3#k魔力，你要把多少点分配给四维?(剩余#r" + freePoint +"#k点)：#b\r\n当前进度：#B0#  0%\r\n", 0, 0, freePoint);
                } else {
                    cm.sendOk("您先把戒指脱下来我才能给你弄呀。");
                    cm.dispose();
                }                
            } else if (selection == 2) {
                if (cm.isQuestCompleted(6003) || cm.isQuestCompleted(21604) ) {
                    if (cm.haveItem(1112303)) {
                        cm.gainItem(1112303, -1);
                        cm.getPlayer().gainEquip(1112304, 8, 8, 8, 8, 2000, 2000, 8, 24, 0, 0, 0, 0, 0, 0, 0, 0, -1);
                        cm.saveOrUpdateCharacterExtendValue("闪耀新星戒指克拉数", "2");
                        cm.sendOk("嗯，看起来你已经完成了您的#b骑宠#k任务！好的，奖励你！");
                        cm.dispose();
                    } else {
                        cm.sendOk("哈哈，不把1克拉戒指先脱下来，是想1克拉2克拉一起戴吗？");
                        cm.dispose();
                    }
                } else {
                    cm.sendOk("您如果是冒险家，执行了#b费洛蒙香水#k这个任务以后再来找我吧，如果是冒险骑士团或者战神，完成获取属于您的#b骑宠#k任务即可。");
                    cm.dispose();
                }
            }
        } else if (RingFlag == 2) {
            if (selection == 0) {
                if (cm.haveItem(1112304)) {
                    freePoint = 40;
                    cm.sendGetNumber("#b闪耀新星戒指2克拉#k一共有#b40点#k可分配属性。每点可分配属性可以变成#b1#k四维、#b250#kHP、#b250#kMP、#b1#k攻击或者#b3#k魔力，你要把多少点分配给四维?(剩余#r" + freePoint + "#k点)：#b\r\n当前进度：#B0#  0%\r\n", 0, 0, freePoint);
                } else {
                    cm.sendOk("您先把戒指脱下来我才能给你弄呀。");
                    cm.dispose();
                }
            } else if (selection == 1) {
                cm.sendOk("再耐心等等，我还在准备。");
                cm.dispose();
            }
        }
    } else if (status == 3) {
        addAbility = selection;
        freePoint -= selection;
        cm.sendGetNumber("#b闪耀新星戒指" + RingFlag + "克拉#k一共有#b" + RingFlag*20 + "点#k可分配属性。每点可分配属性可以变成#b1#k四维、#b250#kHP、#b250#kMP、#b1#k攻击或者#b3#k魔力，你要把多少点分配给HP?(剩余#r" + freePoint +"#k点)：#b\r\n当前进度：#B20#  20%\r\n", 0, 0, freePoint);
    } else if (status == 4) {
        addHp = selection;
        freePoint -= selection;
        cm.sendGetNumber("#b闪耀新星戒指" + RingFlag + "克拉#k一共有#b" + RingFlag * 20 + "点#k可分配属性。每点可分配属性可以变成#b1#k四维、#b250#kHP、#b250#kMP、#b1#k攻击或者#b3#k魔力，你要把多少点分配给MP?(剩余#r" + freePoint + "#k点)：#b\r\n当前进度：#B40#  40%\r\n", 0, 0, freePoint);
    } else if (status == 5) {
        addMp = selection;
        freePoint -= selection;
        cm.sendGetNumber("#b闪耀新星戒指" + RingFlag + "克拉#k一共有#b" + RingFlag * 20 + "点#k可分配属性。每点可分配属性可以变成#b1#k四维、#b250#kHP、#b250#kMP、#b1#k攻击或者#b3#k魔力，你要把多少点分配给攻击?(剩余#r" + freePoint + "#k点)：#b\r\n当前进度：#B60#  60%\r\n", 0, 0, freePoint);
    } else if (status == 6) {
        addATK = selection;
        freePoint -= selection;
        cm.sendGetNumber("#b闪耀新星戒指" + RingFlag + "克拉#k一共有#b" + RingFlag * 20 + "点#k可分配属性。每点可分配属性可以变成#b1#k四维、#b250#kHP、#b250#MP、#b1#k攻击或者#b3#k魔力，你要把多少点分配给魔力?(剩余#r" + freePoint + "#k点)：#b\r\n当前进度：#B80#  80%\r\n", 0, 0, freePoint);
    } else if (status == 7) {
        addMATK = selection;
        freePoint -= selection;
        cm.sendYesNo("#b" + addAbility + "#k四维、#b" + addHp * 250 + "#kHP、#b" + addMp * 250 + "#kMP、#b" + addATK + "#k攻击、#b" + addMATK * 3 + "#k魔力，还剩下#r" + freePoint +"#k点数未分配，对么？\r\n#b当前进度：#B100#  100%\r\n");
    } else if (status == 8) {
        cm.gainItem(1112302+RingFlag, -1);
        cm.getPlayer().gainEquip(1112302+RingFlag, addAbility, addAbility, addAbility, addAbility, addHp * 250, addMp * 250, addATK, addMATK*3, 0, 0, 0, 0, 0, 0, 0, 0, -1);
        cm.sendOk("搞定！");
        cm.dispose();
    }
}
