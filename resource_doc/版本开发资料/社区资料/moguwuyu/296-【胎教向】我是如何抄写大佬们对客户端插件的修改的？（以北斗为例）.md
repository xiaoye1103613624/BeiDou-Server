在技术交流群、论坛等社区里有时会看到大佬发出一两行带着0x00123456的代码，说这串代码实现了某某功能。接着就有人说他不会、他看不懂。然后他会获得的回复大抵是“看某某帖子/视频，去学打包插件”“随便放在哪里就好了”这样的。我写这个帖子是想更优雅地解决此类问题，面向萌新、或者已经是wz编辑与服务端源码修改大佬但是完全不接触客户端逆向的朋友们写。胎教向，抛砖引玉，请大家批评指正。
看完这个帖子可以实现什么
如果你玩的是Beidou GMS083，你可以把别人放出来的客户端逆向修改内容（比如弓镖不挥拳等）加入你自己的客户端中。然后在客户端设置文件里加一个开关，来控制这个新功能的启用或者禁用，比如
[general]
;原有内容
[optional]
;原有内容
;新增开关
;远程职业近战攻击时不会挥拳（设置为true时不会挥拳）
banFist=true
;新增更多开关...
[debug]
;原有内容

//省略原有的内容..............
int Client::talkTime = 2000; // 说话间隔时间

void Client::UpdateGameStartup() {
//省略原有内容..........
	//optional non-resolution related stuff
	if (useTubi) { Memory::FillBytes(0x00485C32, 0x90, 2); }

	//新增的
	if(banFist){ Memory::WriteByte(0x009516C2 + 1, 0x89); }//弓镖不挥拳
	if (StrengthenSewer) {
		//以下为一出双击无视地形
		Memory::WriteByte(0x00950DB9, 0x90);
		Memory::WriteByte(0x00950DBA, 0x90);
		Memory::WriteByte(0x00950DBB, 0x90);
		Memory::WriteByte(0x00950DBC, 0x90);
		Memory::WriteByte(0x00950DBD, 0x90);
		Memory::WriteByte(0x00950DBE, 0x90);
		Memory::WriteByte(0x00950DBF, 0x90);
		Memory::WriteByte(0x00950DC0, 0x90);
		//以上为一出双击无视地形
		unsigned char corkscrewBlow[] = { 0xE9, 0x19, 0x00, 0x00 };
		Memory::WriteByteArray(0x00968278, corkscrewBlow, sizeof(corkscrewBlow));//取消贯骨蓄力
		unsigned char backspinDelay[] = { 0xFE, 0x00 };
		Memory::WriteByteArray(0x00953054 + 1, backspinDelay, sizeof(backspinDelay));//降低回马硬直
		unsigned char  backspinCooltime[] = { 0xB3, 0x04 };
		Memory::WriteByteArray(0x009530FB + 2, backspinCooltime, sizeof(backspinCooltime));//降低回马CD
		unsigned char doubleupper[] = { 0x00, 0x01 };
		Memory::WriteByteArray(0x00953688 + 1, doubleupper, sizeof(doubleupper));//降低升龙内置CD
	}
	if(unStopAttack){ Memory::WriteByte(0x009536E0, 0xEB); }//攻击不停

	Memory::WriteInt(0x0077E055 + 1, 2147483646); // 物攻PAD 相关具体不明，默认值1999，int 4字节
//省略原有的内容...............................
}
