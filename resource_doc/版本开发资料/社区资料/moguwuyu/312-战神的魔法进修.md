**#1楼**

在大多数人的印象里，战神从来不需要魔法。
他只需要一把巨大的长矛、足够强壮的双臂，以及把敌人砸进地里的决心。对他来说，所谓的战斗技巧从来都很简单：冲上去，挥下去，然后确认敌人还会不会动。
直到他遇见了黑魔法师。
那场战斗之后，战神第一次意识到，有些敌人并不是单靠力量就能击败的。长矛可以贯穿铠甲，可以击碎岩石，甚至可以撕开怪物的身体，但面对扭曲空间、腐蚀灵魂、操纵命运的黑暗魔法时，纯粹的蛮力显得有些……不够优雅。
当然，战神本人并不承认这是失败。
他只是冷冷地说道：
“我只是需要更多能砸人的方法。”
于是，在众人震惊的目光中，这位传说中的战士踏进了魔法学院。
魔法学院的导师们一开始并不欢迎他。毕竟，他们很难接受一个学生在练习魔力凝聚时，把法杖当成长矛投出去；也很难接受他在学习元素理论时，坚持认为“火属性魔法”最好的使用方式是把长矛烧红之后砸向敌人。
但出乎意料的是，战神并不是完全没有魔法天赋。
他只是理解魔法的方式和别人不太一样。
别人学习魔法，是为了引导元素、构筑术式、释放咒文。
而战神学习魔法，是为了让自己的每一次攻击都拥有第二次回响。
当他的长矛划破空气时，残留的魔力会在轨迹中再次凝聚，像是战斗本身不愿意结束一样，在下一瞬间重新爆发。有人说那是魔法回声，也有人说那是战神的斗志在空间中留下的影子。
学院最终不得不承认：虽然他的施法方式毫无理论美感，动作粗暴，咒文基本靠吼，但效果确实非常惊人。
从那以后，战神依旧不喜欢自称为魔法师。
他只是承认了一件事：
面对黑魔法师那样的敌人，光靠力量还不够。
所以，他决定把魔法也变成力量的一部分。
*.h
#pragma once

#include <Windows.h>
#include <iostream>

class FinalBlowPro
{
public:
    static void Install();
    static void Uninstall();

private:
    static constexpr int kFinalBlowSkillId = 21120005;
    static constexpr int kAran4JobId = 2112;
    static constexpr int kTestKey = 'W';

    static constexpr DWORD kCUserLocal_OnKey = 0x0094C856;

    static constexpr DWORD kCUserLocal_DoActiveSkill = 0x00966F7A;

private:
    using OnKey_t = void(__thiscall*)(void* pThis, int nKey, int nScanCode);
    using DoActiveSkill_t = int(__thiscall*)(void* pThis, int nSkillID, unsigned int nScanCode, int* pnConsumeCheck);
    using GetJob_t = int(__thiscall*)(void* pThis);

private:
    static OnKey_t s_OnKey;
    static DoActiveSkill_t s_DoActiveSkill;

    static void __fastcall OnKey_Hook(void* pThis, void* edx, int nKey, int nScanCode);

    static int GetJobId(void* pUser);
    static bool IsKeyDown(int nScanCode);
    static bool CanTrigger();
};
*.cpp
#include "FinalBlowPro.h"
#include <detours.h>

#pragma comment(lib, "detours.lib")

FinalBlowPro::OnKey_t FinalBlowPro::s_OnKey = nullptr;
FinalBlowPro::DoActiveSkill_t FinalBlowPro::s_DoActiveSkill = nullptr;

void FinalBlowPro::Install()
{
    if (kCUserLocal_OnKey == 0 || kCUserLocal_DoActiveSkill == 0)
    {
        std::cout << "[FinalBlowPro] Install failed: address is not set." << std::endl;
        return;
    }

    s_OnKey = reinterpret_cast<OnKey_t>(kCUserLocal_OnKey);
    s_DoActiveSkill = reinterpret_cast<DoActiveSkill_t>(kCUserLocal_DoActiveSkill);

    DetourTransactionBegin();
    DetourUpdateThread(GetCurrentThread());
    DetourAttach(reinterpret_cast<PVOID*>(&s_OnKey), OnKey_Hook);
    LONG ret = DetourTransactionCommit();

    if (ret == NO_ERROR)
    {
        std::cout << "[FinalBlowPro] Install success." << std::endl;
    }
    else
    {
        std::cout << "[FinalBlowPro] Install failed, Detour error = " << ret << std::endl;
    }
}

void FinalBlowPro::Uninstall()
{
    if (!s_OnKey)
        return;

    DetourTransactionBegin();
    DetourUpdateThread(GetCurrentThread());
    DetourDetach(reinterpret_cast<PVOID*>(&s_OnKey), OnKey_Hook);
    LONG ret = DetourTransactionCommit();

    if (ret == NO_ERROR)
    {
        std::cout << "[FinalBlowPro] Uninstall success." << std::endl;
    }
    else
    {
        std::cout << "[FinalBlowPro] Uninstall failed, Detour error = " << ret << std::endl;
    }
}

void __fastcall FinalBlowPro::OnKey_Hook(void* pThis, void* edx, int nKey, int nScanCode)
{
    if (!pThis || !s_DoActiveSkill)
    {
        s_OnKey(pThis, nKey, nScanCode);
        return;
    }

    if (IsKeyDown(nScanCode) && nKey == kTestKey)
    {
        const int jobId = GetJobId(pThis);

        if (jobId == kAran4JobId && CanTrigger())
        {
            int consumeCheck = 0;

            const int result = s_DoActiveSkill(
                pThis,
                kFinalBlowSkillId,
                static_cast<unsigned int>(nScanCode),
                &consumeCheck
            );

            std::cout << "[FinalBlowPro] W pressed. "
                << "jobId = " << jobId
                << ", skillId = " << kFinalBlowSkillId
                << ", result = " << result
                << ", consumeCheck = " << consumeCheck
                << std::endl;       
            return;
        }
    }

    s_OnKey(pThis, nKey, nScanCode);
}

int FinalBlowPro::GetJobId(void* pUser)
{
    if (!pUser)
        return 0;
    DWORD* vtbl = *reinterpret_cast<DWORD**>(pUser);

    if (!vtbl)
        return 0;

    GetJob_t fnGetJob = reinterpret_cast<GetJob_t>(vtbl[16]);

    if (!fnGetJob)
        return 0;

    return fnGetJob(pUser);
}

bool FinalBlowPro::IsKeyDown(int nScanCode)
{   
    return nScanCode >= 0;
}

bool FinalBlowPro::CanTrigger()
{
    static DWORD s_lastTriggerTime = 0;

    DWORD now = GetTickCount();

    // 防止按住 W 自动连发
    if (now - s_lastTriggerTime < 250)
        return false;

    s_lastTriggerTime = now;
    return true;
}
