# 不稳定魔法弹

> 来源：https://moguwuyu.com/d/308
> 站点：蘑菇物语(moguwuyu.com) · Flarum 改端技术论坛

**#1楼**

在魔法学院的入门课程中，所有新手魔法师都被要求掌握一项最基础的法术：魔法弹。
理论上，它应该从施法者的法杖尖端出发，以一条稳定、优雅、可预测的直线命中目标。 这也是导师们最喜欢它的原因：简单，安全，容易评分。
然而，总有一些学生对“稳定”二字缺乏基本尊重。
据说第一位施放出“不稳定魔法弹”的新手魔法师，并不是出于天赋，也不是出于研究精神，而是在念咒时不小心把“凝聚魔力”念成了“随便飞吧”。从那以后，他的魔法弹便拒绝遵守任何几何学常识。
它会旋转、漂移、抽搐、绕弯，有时像喝醉的萤火虫，有时像被风吹走的蓝色泡泡。旁观者往往会误以为施法失败，直到那颗摇摇晃晃的魔法弹在最后一刻忽然想起自己的职责，并精准地砸在怪物脸上。
学院导师最初试图禁止这种法术，理由是它“缺乏可重复性、缺乏稳定性、缺乏尊严”。
但实战记录显示，不稳定魔法弹虽然飞行路线荒唐，却总能以一种令人难以理解的方式抵达目标。
于是，魔法学院只好把它写进教材的附录中，并在旁边加上一行小字：
"不建议模仿。除非你运气很好"
如今，许多新手魔法师都偷偷练习这门法术。
他们坚称，这不是失误，而是一种高级弹道扰动技术。
至于为什么魔法弹会在空中像粒子一样随机飞舞，又为什么最后仍然能命中目标——
大多数魔法师认为，这是复杂的奥术轨迹计算。
而少数诚实的新手则承认：
“我也不知道，反正它最后打中了。”

    struct MagicBulletNode
    {
        void* vtable;        // +0x00
        void* layer;         // +0x04，创建后的 IWzGr2DLayer*
        int flagOrState;     // +0x08
        int unknown0C;       // +0x0C
        int unknown10;       // +0x10
        int unknown14;       // +0x14

        int startTime;       // +0x18
        int endTime;         // +0x1C

        int x1;              // +0x20 起点 X
        int y1;              // +0x24 起点 Y

        int y2;              // +0x2C 终点 Y

    bool IsBadPtrSafe(void* p)
    {
        if (!p)
            return true;

        MEMORY_BASIC_INFORMATION mbi{};
        if (!VirtualQuery(p, &mbi, sizeof(mbi)))
            return true;

        if (mbi.State != MEM_COMMIT)
            return true;

        if (mbi.Protect & PAGE_NOACCESS)
            return true;

        if (mbi.Protect & PAGE_GUARD)
            return true;

        return false;
    }

    int FastRandRange(int minValue, int maxValue)
    {
        static unsigned int seed = GetTickCount();

        seed = seed * 1103515245 + 12345;

        const int range = maxValue - minValue + 1;
        if (range <= 0)
            return minValue;

        return minValue + static_cast<int>((seed >> 16) % range);
    }

    float ClampFloat(float value, float minValue, float maxValue)
    {
        if (value < minValue)
            return minValue;

        if (value > maxValue)
            return maxValue;

        return value;
    }

    int ClampInt(int value, int minValue, int maxValue)
    {
        if (value < minValue)
            return minValue;

        if (value > maxValue)
            return maxValue;

        return value;
    }

    void PrintNode(const char* tag, MagicBulletNode* node, unsigned int tCur)
    {
        if (!kLogEnabled)
            return;

        if (!node || IsBadPtrSafe(node))
            return;

        DWORD vt = reinterpret_cast<DWORD>(node->vtable);
        DWORD updateFn = 0;

        if (node->vtable && !IsBadPtrSafe(node->vtable))
        {
            updateFn = *reinterpret_cast<DWORD*>(
                reinterpret_cast<char*>(node->vtable) + 4
                );
        }

        std::cout
            << "[" << tag << "] "
            << "tCur=" << tCur
            << " node=0x" << std::hex << reinterpret_cast<DWORD>(node)
            << " vt=0x" << vt
            << " update=0x" << updateFn
            << " layer=0x" << reinterpret_cast<DWORD>(node->layer)
            << " origin=0x" << reinterpret_cast<DWORD>(node->origin)
            << " bstr=0x" << reinterpret_cast<DWORD>(node->bstrData)
            << std::dec
            << " startTime=" << node->startTime
            << " endTime=" << node->endTime
            << " p1=(" << node->x1 << "," << node->y1 << ")"
            << " p2=(" << node->x2 << "," << node->y2 << ")"
            << " a8=" << node->a8
            << " a10=" << node->a10
            << std::endl;
    }

    void AdjustNodeFlightTimeBeforeCreateLayer(MagicBulletNode* node)
    {
        if (!kModifyOriginalFlightTime)
            return;

        if (!node || IsBadPtrSafe(node))
            return;

        const int startTime = node->startTime;
        const int oldEndTime = node->endTime;

        int oldDuration = oldEndTime - startTime;
        if (oldDuration <= 0)
            return;

        int newDuration = static_cast<int>(
            static_cast<float>(oldDuration) * kOriginalFlightTimeScale
            );

        newDuration = ClampInt(
            newDuration,
            kMinFlightDurationMs,
            kMaxFlightDurationMs
        );

        node->endTime = startTime + newDuration;

        if (kLogEnabled)
        {
            std::cout
                << "[MagicBullet TIME_ADJUST] "
                << "node=0x" << std::hex << reinterpret_cast<DWORD>(node)
                << std::dec
                << " startTime=" << startTime
                << " oldEndTime=" << oldEndTime
                << " newEndTime=" << node->endTime
                << " oldDuration=" << oldDuration
                << " newDuration=" << newDuration
                << " scale=" << kOriginalFlightTimeScale
                << std::endl;
        }
    }

    Vec2i MakeCurveControlPoint(int sx, int sy, int ex, int ey)
    {
        Vec2i cp{};

        const int dx = ex - sx;
        const int dy = ey - sy;

        const int midX = (sx + ex) / 2;
        const int midY = (sy + ey) / 2;

        int perpX = -dy;
        int perpY = dx;

        const int absX = perpX >= 0 ? perpX : -perpX;
        const int absY = perpY >= 0 ? perpY : -perpY;

        int approxLen = absX + absY;
        if (approxLen <= 0)
            approxLen = 1;

        int curvePower = FastRandRange(kCurveMinPower, kCurveMaxPower);

        if (FastRandRange(0, 1) == 0)
            curvePower = -curvePower;

        cp.x = midX + perpX * curvePower / approxLen;
        cp.y = midY + perpY * curvePower / approxLen;

        cp.x += FastRandRange(-kControlJitter, kControlJitter);
        cp.y += FastRandRange(-kControlJitter, kControlJitter);

        return cp;
    }

    Vec2i Bezier2(
        int x0,
        int y0,
        int x1,
        int y1,
        int x2,
        int y2,
        float t
    )
    {
        const float u = 1.0f - t;

        Vec2i p{};

        p.x = static_cast<int>(
            u * u * static_cast<float>(x0) +
            2.0f * u * t * static_cast<float>(x1) +
            t * t * static_cast<float>(x2)
            );

        p.y = static_cast<int>(
            u * u * static_cast<float>(y0) +
            2.0f * u * t * static_cast<float>(y1) +
            t * t * static_cast<float>(y2)
            );

        return p;
    }

    Vec2i LerpPoint(int x0, int y0, int x1, int y1, float t)
    {
        t = ClampFloat(t, 0.0f, 1.0f);

        Vec2i p{};
        p.x = static_cast<int>(
            static_cast<float>(x0) * (1.0f - t) +
            static_cast<float>(x1) * t
            );

        p.y = static_cast<int>(
            static_cast<float>(y0) * (1.0f - t) +
            static_cast<float>(y1) * t
            );

        return p;
    }

    Vec2i MakeOvershootEndPoint(int sx, int sy, int ex, int ey, int distance)
    {
        Vec2i p{};
        p.x = ex;
        p.y = ey;

        const int dx = ex - sx;
        const int dy = ey - sy;

        const int absX = dx >= 0 ? dx : -dx;
        const int absY = dy >= 0 ? dy : -dy;

        int approxLen = absX + absY;
        if (approxLen <= 0)
            approxLen = 1;

        p.x = ex + dx * distance / approxLen;
        p.y = ey + dy * distance / approxLen;

        return p;
    }

    float SmoothStep(float t)
    {
        t = ClampFloat(t, 0.0f, 1.0f);
        return t * t * (3.0f - 2.0f * t);
    }

    int TriangleWave01(int phase)
    {
        phase %= 1024;

        if (phase < 0)
            phase += 1024;

        if (phase < 512)
            return phase;

        return 1024 - phase;
    }

    int SignedTriangleWave(int phase, int amp)
    {
        int v = TriangleWave01(phase);
        v = v - 256;
        return v * amp / 256;
    }

    bool SetLayerRxRy(void* layer, int x, int y)
    {
        if (!layer)
            return false;

        if (IsBadPtrSafe(layer))
            return false;

        auto* vec = reinterpret_cast<IWzVector2D*>(layer);

        HRESULT hrX = vec->put_rx(x);
        HRESULT hrY = vec->put_ry(y);

        return SUCCEEDED(hrX) && SUCCEEDED(hrY);
    }

    bool SetLayerRelMoveNow(void* layer, int x, int y)
    {
        if (!layer)
            return false;

        if (IsBadPtrSafe(layer))
            return false;

        auto* vec = reinterpret_cast<IWzVector2D*>(layer);

        VARIANT vtEmpty1;
        VARIANT vtEmpty2;
        VariantInit(&vtEmpty1);
        VariantInit(&vtEmpty2);

        HRESULT hr = vec->raw_RelMove(x, y, vtEmpty1, vtEmpty2);

        return SUCCEEDED(hr);
    }

    void RegisterCurveRuntimeIfNeeded(MagicBulletNode* node)
    {
        if (!node)
            return;

        if (IsBadPtrSafe(node))
            return;

        auto& rt = g_curveRuntime[node];

        if (rt.initialized)
            return;

        rt.initialized = true;

        rt.startX = node->x1;
        rt.startY = node->y1;

        rt.endX = node->x2;
        rt.endY = node->y2;

        rt.startTime = static_cast<unsigned int>(node->startTime);
        rt.endTime = static_cast<unsigned int>(node->endTime);

        Vec2i cp = MakeCurveControlPoint(
            rt.startX,
            rt.startY,
            rt.endX,
            rt.endY
        );

        rt.controlX = cp.x;
        rt.controlY = cp.y;

        rt.wiggleSeed = FastRandRange(0, 4096);

        if (kTrajectoryMode == 0)
        {
            rt.trajectoryMode = FastRandRange(1, 4);
        }
        else
        {
            rt.trajectoryMode = kTrajectoryMode;
        }

        if (kLogEnabled)
        {
            std::cout
                << "[MagicBullet CURVE_REGISTER] "
                << "node=0x" << std::hex << reinterpret_cast<DWORD>(node)
                << std::dec
                << " mode=" << rt.trajectoryMode
                << " start=(" << rt.startX << "," << rt.startY << ")"
                << " control=(" << rt.controlX << "," << rt.controlY << ")"
                << " end=(" << rt.endX << "," << rt.endY << ")"
                << " startTime=" << rt.startTime
                << " endTime=" << rt.endTime
                << " seed=" << rt.wiggleSeed
                << std::endl;
        }
    }

    Vec2i BuildTrajectoryPoint(const CurveRuntime& rt, float rawT)
    {
        float visualT = rawT * kBulletSpeedScale;
        visualT = ClampFloat(visualT, 0.0f, 1.0f);

        visualT = SmoothStep(visualT);

        Vec2i pos = Bezier2(
            rt.startX,
            rt.startY,
            rt.controlX,
            rt.controlY,
            rt.endX,
            rt.endY,
            visualT
        );

        const int dx = rt.endX - rt.startX;
        const int dy = rt.endY - rt.startY;

        int perpX = -dy;
        int perpY = dx;

        const int absX = perpX >= 0 ? perpX : -perpX;
        const int absY = perpY >= 0 ? perpY : -perpY;

        int approxLen = absX + absY;
        if (approxLen <= 0)
            approxLen = 1;

        const int phase = static_cast<int>(visualT * 1024.0f);

        if (rt.trajectoryMode == 1)
        {
            int wave = SignedTriangleWave(
                phase * kSWaveCount + rt.wiggleSeed,
                kSWaveAmplitude
            );

            float envelopeF = 4.0f * visualT * (1.0f - visualT);
            envelopeF = ClampFloat(envelopeF, 0.0f, 1.0f);

            wave = static_cast<int>(static_cast<float>(wave) * envelopeF);

            pos.x += perpX * wave / approxLen;
            pos.y += perpY * wave / approxLen;
        }
        else if (rt.trajectoryMode == 2)
        {
            float envelopeF = 4.0f * visualT * (1.0f - visualT);
            envelopeF = ClampFloat(envelopeF, 0.0f, 1.0f);

            int waveNormal = SignedTriangleWave(
                phase * kSpiralTurns + rt.wiggleSeed,
                kSpiralAmplitude
            );

            int waveAlong = SignedTriangleWave(
                phase * kSpiralTurns + rt.wiggleSeed + 256,
                kSpiralAmplitude / 2
            );

            waveNormal = static_cast<int>(static_cast<float>(waveNormal) * envelopeF);
            waveAlong = static_cast<int>(static_cast<float>(waveAlong) * envelopeF);

            pos.x += perpX * waveNormal / approxLen;
            pos.y += perpY * waveNormal / approxLen;

            pos.x += dx * waveAlong / approxLen;
            pos.y += dy * waveAlong / approxLen;
        }
        else if (rt.trajectoryMode == 3)
        {
            Vec2i over = MakeOvershootEndPoint(
                rt.startX,
                rt.startY,
                rt.endX,
                rt.endY,
                kOvershootDistance
            );

            if (visualT < kOvershootSplitT)
            {
                float t1 = visualT / kOvershootSplitT;
                t1 = SmoothStep(t1);

                pos = Bezier2(
                    rt.startX,
                    rt.startY,
                    rt.controlX,
                    rt.controlY,
                    over.x,
                    over.y,
                    t1
                );
            }
            else
            {
                float t2 = (visualT - kOvershootSplitT) / (1.0f - kOvershootSplitT);
                t2 = SmoothStep(t2);

                pos = LerpPoint(
                    over.x,
                    over.y,
                    rt.endX,
                    rt.endY,
                    t2
                );
            }
        }
        else
        {
            Vec2i over = MakeOvershootEndPoint(
                rt.startX,
                rt.startY,
                rt.endX,
                rt.endY,
                kOvershootDistance
            );

            if (visualT < kOvershootSplitT)
            {
                float t1 = visualT / kOvershootSplitT;
                t1 = SmoothStep(t1);

                pos = Bezier2(
                    rt.startX,
                    rt.startY,
                    rt.controlX,
                    rt.controlY,
                    over.x,
                    over.y,
                    t1
                );

                float env1 = 4.0f * t1 * (1.0f - t1);
                env1 = ClampFloat(env1, 0.0f, 1.0f);

                int wave = SignedTriangleWave(
                    static_cast<int>(t1 * 1024.0f) * kSWaveCount + rt.wiggleSeed,
                    kSWaveAmplitude
                );

                wave = static_cast<int>(static_cast<float>(wave) * env1);

                pos.x += perpX * wave / approxLen;
                pos.y += perpY * wave / approxLen;
            }
            else
            {
                float t2 = (visualT - kOvershootSplitT) / (1.0f - kOvershootSplitT);
                t2 = SmoothStep(t2);

                pos = LerpPoint(
                    over.x,
                    over.y,
                    rt.endX,
                    rt.endY,
                    t2
                );

                float env2 = 1.0f - t2;
                env2 = ClampFloat(env2, 0.0f, 1.0f);

                int smallWave = SignedTriangleWave(
                    static_cast<int>(t2 * 1024.0f) * 3 + rt.wiggleSeed,
                    kSWaveAmplitude / 3
                );

                smallWave = static_cast<int>(static_cast<float>(smallWave) * env2);

                pos.x += perpX * smallWave / approxLen;
                pos.y += perpY * smallWave / approxLen;
            }
        }

        // 最后强制收束到原始终点
        if (rawT > 0.94f)
        {
            float k = (rawT - 0.94f) / 0.06f;
            k = ClampFloat(k, 0.0f, 1.0f);

            pos.x = static_cast<int>(
                static_cast<float>(pos.x) * (1.0f - k) +
                static_cast<float>(rt.endX) * k
                );

            pos.y = static_cast<int>(
                static_cast<float>(pos.y) * (1.0f - k) +
                static_cast<float>(rt.endY) * k
                );
        }

        return pos;
    }

    void UpdateCurveLayerPosition(MagicBulletNode* node, unsigned int tCur)
    {
        if (!kEnableCurveMove)
            return;

        if (!node)
            return;

        if (IsBadPtrSafe(node))
            return;

        if (!node->layer)
            return;

        auto it = g_curveRuntime.find(node);
        if (it == g_curveRuntime.end())
            return;

        CurveRuntime& rt = it->second;

        if (!rt.initialized)
            return;

        const int duration = static_cast<int>(rt.endTime - rt.startTime);
        if (duration <= 0)
            return;

        float rawT = static_cast<float>(
            static_cast<int>(tCur - rt.startTime)
            ) / static_cast<float>(duration);

        rawT = ClampFloat(rawT, 0.0f, 1.0f);

        Vec2i pos = BuildTrajectoryPoint(rt, rawT);

        const bool moved = MoveLayerToCurvePos(node->layer, pos.x, pos.y);

        if (kLogCurveMove && !rt.printedMiddle && rawT > 0.40f)
        {
            rt.printedMiddle = true;

            std::cout
                << "[MagicBullet CURVE_MOVE] "
                << "node=0x" << std::hex << reinterpret_cast<DWORD>(node)
                << " layer=0x" << reinterpret_cast<DWORD>(node->layer)
                << std::dec
                << " mode=" << rt.trajectoryMode
                << " rawT=" << rawT
                << " pos=(" << pos.x << "," << pos.y << ")"
                << " moved=" << (moved ? "true" : "false")
                << std::endl;
        }
    }

    // ============================================================
    // MagicBullet Node Hook
    // ============================================================

    void PatchNodeBeforeCreateLayer(MagicBulletNode* node, unsigned int tCur)
    {
        if (!node)
            return;

        if (IsBadPtrSafe(node))
            return;

        if (node->layer != nullptr)
            return;

        if (g_patchedNodes.find(node) != g_patchedNodes.end())
            return;

        if (g_patchedNodes.size() > 1024)
        {
            g_patchedNodes.clear();
            g_curveRuntime.clear();

            if (kLogEnabled)
            {
                std::cout << "[MagicBullet] cache auto cleared" << std::endl;
            }
        }

        g_patchedNodes.insert(node);

        PrintNode("MagicBullet BEFORE", node, tCur);

        // 关键：先调整原始飞行时间，再注册曲线数据
        AdjustNodeFlightTimeBeforeCreateLayer(node);

        RegisterCurveRuntimeIfNeeded(node);

        if (kLogEnabled)
        {
            std::cout
                << "[MagicBullet KEEP_ENDPOINT] "
                << "p1=(" << node->x1 << "," << node->y1 << ") "
                << "p2=(" << node->x2 << "," << node->y2 << ") "
                << "endpoint unchanged"
                << std::endl;
        }
    }

    int __fastcall hkMagicBulletNodeUpdate(MagicBulletNode* node, void* edx, unsigned int tCur)
    {
        PatchNodeBeforeCreateLayer(node, tCur);

        int ret = oMagicBulletNodeUpdate(node, tCur);

        if (!ret && node && !IsBadPtrSafe(node))
        {
            UpdateCurveLayerPosition(node, tCur);
        }

        if (ret && node)
        {
            g_patchedNodes.erase(node);
            g_curveRuntime.erase(node);

            if (kLogEnabled)
            {
                std::cout
                    << "[MagicBullet END] "
                    << "node=0x" << std::hex << reinterpret_cast<DWORD>(node)
                    << std::dec
                    << " removed from cache"
                    << std::endl;
            }
        }

        return ret;
    }

    // ============================================================
    // Install
    // ============================================================

    void Install()
    {
        oMagicBulletNodeUpdate =
            reinterpret_cast<MagicBulletNodeUpdate_t>(kMagicBulletNodeUpdateAddr);

        ATTACH_HOOK(oMagicBulletNodeUpdate, hkMagicBulletNodeUpdate);

        std::cout
            << "[MagicBulletHook] NodeUpdate installed at 0x"
            << std::hex << kMagicBulletNodeUpdateAddr
            << std::dec
            << std::endl;
    }
